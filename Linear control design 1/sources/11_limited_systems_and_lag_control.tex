\section{Limited systems and Lag control}

\subsection{Rate and amplitude limitations}

\subsubsection{Amplitude saturation}
\[
u_{\mathrm{sat}}(t)=
\begin{cases}
u_{\max}, & u(t)>u_{\max},\\
u(t), & u_{\min}\leq u(t)\leq u_{\max},\\
u_{\min}, & u(t)<u_{\min}
\end{cases}
\]
Amplitude saturation is nonlinear and can invalidate a linear controller design.

\subsubsection{Rate limitation}
\[
|\dot{u}(t)|\leq R_{\max}
\]
Rate limits restrict how quickly the input can change and can create slower or unstable closed-loop behavior.

\subsubsection{Integrator windup risk}
\[
u_I(t)=\frac{K_P}{\tau_i}\int_0^t e(\xi)\,d\xi
\]
When the actuator saturates, the integral state can keep growing although the physical input cannot follow.

\subsection{Mitigation strategies}

\subsubsection{Anti-windup back-calculation}
\[
\dot{x}_I=e+\frac{1}{T_t}\bigl(u_{\mathrm{sat}}-u\bigr)
\]
Back-calculation reduces integral buildup when saturation separates requested input from applied input. % TODO: verify from source

\subsubsection{Reference filtering for rate limits}
\[
F_r(s)=\frac{1}{\tau_f s+1}
\]
A prefilter can slow reference changes so that the commanded input respects rate limitations.

\subsection{P-Lead-Lag control}

\subsubsection{Lag transfer function}
\[
C_{\mathrm{lag}}(s)=\frac{\tau_i s+1}{\frac{\tau_i}{\beta}s+1},\qquad \beta>1
\]
The lag part increases low-frequency gain relative to high-frequency gain without adding an integrator.

\subsubsection{P-Lead-Lag transfer function}
\[
C(s)=K_P
\frac{\tau_d s+1}{\alpha\tau_d s+1}
\frac{\tau_i s+1}{\frac{\tau_i}{\beta}s+1}
\]
P-Lead-Lag design can improve low-frequency accuracy while avoiding unlimited integral buildup.

\subsubsection{Lag phase contribution}
\[
\phi_L(\omega_c)=
\arctan\left(\frac{N_i(1-\beta)}{1+\beta N_i^2}\right),
\qquad N_i=\omega_c\tau_i
\]
This expression appears in exam and slide design tasks for selecting the lag parameter \(\beta\).

\subsubsection{Non-zero steady-state error with Lag}
\[
e_{\mathrm{ss}}=\frac{1}{1+L(0)}
\]
A lag controller gives finite low-frequency gain, so zero step error is not guaranteed.
