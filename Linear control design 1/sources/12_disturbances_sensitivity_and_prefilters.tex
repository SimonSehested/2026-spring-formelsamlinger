\section{Disturbances, sensitivity, and prefilters}

\subsection{Ctrl+F keywords}

\subsubsection{Disturbance, forstyrrelse, sensitivity, prefilter, measurement noise, maaling}
\[
\text{output disturbance, input disturbance, }S,T,G_{yd},G_{ur},F(s),\text{ noise}
\]
Use this section when the problem asks how disturbances, noise, or prefilters affect the closed loop.

\subsection{Sensitivity functions}

\subsubsection{Sensitivity and complementary sensitivity}
\[
S(s)=\frac{1}{1+L(s)},\qquad
T(s)=\frac{L(s)}{1+L(s)}
\]
\(S\) maps output disturbances and tracking error. \(T\) maps reference tracking and measurement-noise transmission.

\subsubsection{Input sensitivity}
\[
G_{ur}(s)=\frac{C(s)}{1+C(s)G(s)}
\]
Use this to estimate control effort and actuator stress.

\subsection{Disturbance paths}

\subsubsection{Output disturbance after plant}
\[
Y(s)=\frac{G(s)C(s)}{1+C(s)G(s)}R(s)
+\frac{1}{1+C(s)G(s)}D(s)
\]
Additive output disturbances are attenuated by \(S(s)\).

\subsubsection{Input disturbance before plant}
\[
Y(s)=\frac{G(s)C(s)}{1+C(s)G(s)}R(s)
+\frac{G(s)}{1+C(s)G(s)}D(s)
\]
The plant shapes the disturbance before feedback attenuation.

\subsubsection{Measurement noise}
\[
Y(s)=\frac{G(s)C(s)}{1+C(s)G(s)}R(s)
-\frac{G(s)C(s)}{1+C(s)G(s)}N(s)
\]
Measurement noise is transmitted through the complementary sensitivity path.

\subsection{Prefilters}

\subsubsection{Reference prefilter}
\[
Y(s)=\frac{F(s)L(s)}{1+L(s)}R(s)
\]
A prefilter shapes reference response without changing \(L(s)\), stability margins, or disturbance rejection.

\subsubsection{First-order prefilter}
\[
F(s)=\frac{1}{\tau_f s+1}
\]
Low-pass prefilters reduce fast reference commands and actuator peaks.

\subsection{Exam recipe: derive disturbance transfer function}

\textbf{Use when:} the block diagram includes a disturbance input \(d\) or noise \(n\).

\textbf{Steps:}
\begin{enumerate}
\item Set all other inputs to zero by superposition.
\item Write the summing-junction equations.
\item Solve for \(Y(s)/D(s)\), \(E(s)/D(s)\), or the requested signal.
\item Express the result using \(S=1/(1+L)\) and \(T=L/(1+L)\) when possible.
\item Use \(s\to0\) for constant disturbance checks.
\end{enumerate}

\textbf{Fast checks:} output disturbances usually contain \(S\); measurement noise usually contains \(T\).

\textbf{Common traps:} treating input and output disturbances as the same transfer path.

\subsection{Exam recipe: steady-state disturbance rejection}

\textbf{Use when:} the task asks for final output or error due to a constant disturbance.

\textbf{Steps:}
\begin{enumerate}
\item Find the disturbance-to-output or disturbance-to-error transfer function.
\item Insert \(D(s)=d_0/s\).
\item Check closed-loop stability.
\item Compute the final value with \(\lim_{s\to0}s(\cdot)\).
\end{enumerate}

\textbf{Fast checks:} high low-frequency loop gain makes \(S(0)\) small and improves output-disturbance rejection.

\subsection{Exam recipe: prefilter effect}

\textbf{Use when:} a task asks whether a prefilter changes margins or disturbance rejection.

\textbf{Steps:}
\begin{enumerate}
\item Check whether \(F(s)\) is outside the feedback loop.
\item If outside, it changes \(R\to Y\) but not \(L(s)\).
\item Stability margins, \(S\), and disturbance rejection are unchanged.
\item Reference speed and overshoot can change.
\end{enumerate}

\textbf{Common traps:} redesigning \(K_P\) because of a prefilter; the loop transfer is unchanged.
