\section{Design specifications and tuning}

\subsection{Ctrl+F keywords}

\subsubsection{Steady-state error, stationaer fejl, system type, tuning, specification}
\[
\text{step error, ramp error, }K_p,K_v,K_a,\omega_c,\gamma_M,M_p,t_s,K_u,T_u
\]
Use this section when the task gives a performance specification instead of a controller formula.

\subsection{Steady-state error}

\subsubsection{Final value error formula}
\[
e_{\mathrm{ss}}=\lim_{t\to\infty}e(t)=\lim_{s\to0}sE(s)
\]
The closed loop must be stable before this formula is valid.

\subsubsection{Unity-feedback step error}
\[
e_{\mathrm{ss}}=\lim_{s\to0}\frac{1}{1+L(s)}
\]
For a unit step and unity feedback, low-frequency loop gain directly sets tracking error.

\subsubsection{System type and error constants}
\[
L(s)=\frac{K\prod_i(s-z_i)}{s^q\prod_k(s-p_k)}
\]
\[
K_p=\lim_{s\to0}L(s),\qquad K_v=\lim_{s\to0}sL(s),\qquad K_a=\lim_{s\to0}s^2L(s)
\]
The system type \(q\) is the number of loop integrators.

\subsubsection{Reference error table}
\[
\begin{array}{c|ccc}
\text{Type} & \text{step} & \text{ramp} & \text{parabolic}\\ \hline
0 & 1/(1+K_p) & \infty & \infty\\
1 & 0 & 1/K_v & \infty\\
2 & 0 & 0 & 1/K_a
\end{array}
\]
Integral action raises system type and improves polynomial reference tracking.

\subsection{Frequency specifications}

\subsubsection{Crossover speed rule}
\[
\omega_c\uparrow\quad\Longrightarrow\quad t_r\downarrow
\]
Higher crossover usually gives faster tracking, until limited by phase margin, noise, saturation, and unmodelled dynamics.

\subsubsection{Phase margin damping rule}
\[
\gamma_M\uparrow\quad\Longrightarrow\quad M_p\downarrow
\]
Larger phase margin normally means more damping and lower overshoot.

\subsection{P-controller tuning from stability boundary}

\subsubsection{Critical proportional gain}
\[
|K_PG(j\omega_\pi)|=1,\qquad \angle G(j\omega_\pi)=-180^\circ
\]
At the stability boundary, the loop crosses the critical point.

\subsubsection{Magnitude conversion for tuning}
\[
M_{\mathrm{dB}}=20\log_{10}M,\qquad M=10^{M_{\mathrm{dB}}/20}
\]
Use this conversion when reading Bode plots and choosing controller gains.

\subsection{Exam recipe: steady-state error from diagram or Bode DC gain}

\textbf{Use when:} the problem asks for final tracking error, static error, or DC sensitivity.

\textbf{Steps:}
\begin{enumerate}
\item Derive \(E(s)/R(s)\) or use \(S(s)=1/(1+L(s))\) for unity feedback.
\item Insert \(R(s)=1/s\) for a unit step.
\item Check closed-loop stability.
\item Compute \(e_{\mathrm{ss}}=\lim_{s\to0}sE(s)\).
\item If only a Bode plot is given, convert low-frequency dB gain to linear gain first.
\end{enumerate}

\textbf{Fast checks:} type 0 gives finite nonzero step error; type 1 gives zero step error if stable.

\textbf{Common traps:} using the reference-error table for non-unity feedback without deriving the actual \(E/R\).

\subsection{Exam recipe: select answer from design specifications}

\textbf{Use when:} answer options differ by speed, overshoot, or stability margins.

\textbf{Steps:}
\begin{enumerate}
\item Translate time-domain wording to \(M_p,t_s,t_r,e_{\mathrm{ss}}\).
\item Translate frequency-domain wording to \(\omega_c,\gamma_M,A_M\).
\item Use \(\omega_c\) for speed and \(\gamma_M\) for damping.
\item Reject options with unstable poles, negative time constants, or impossible margins.
\end{enumerate}

\textbf{Fast checks:} high \(K_P\) normally increases speed but reduces phase margin.
