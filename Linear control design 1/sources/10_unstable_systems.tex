\section{Unstable systems}

\subsection{Ctrl+F keywords}

\subsubsection{Unstable system, open-loop unstable, right-half-plane pole, ustabil}
\[
\text{RHP pole, stabilization, inner loop, outer loop, nested loop, Nyquist}
\]
Use this section when the plant has a pole with positive real part or the task warns that the open loop is unstable.

\subsection{Unstable plant checks}

\subsubsection{Open-loop unstable pole}
\[
\exists p_k:\operatorname{Re}(p_k)>0
\quad\Longrightarrow\quad
G(s)\ \text{is open-loop unstable}
\]
Stable-plant Bode margin shortcuts do not directly decide closed-loop stability.

\subsubsection{Closed-loop stabilization target}
\[
T(s)=\frac{C(s)G(s)}{1+C(s)G(s)},\qquad
1+C(s)G(s)=0
\]
The controller must make all roots of the closed-loop characteristic equation lie in the open left half-plane.

\subsubsection{Nyquist requirement}
\[
Z=P+N=0
\]
For an open-loop unstable plant, encirclements must compensate the \(P\) right-half-plane open-loop poles. % TODO: verify sign convention from source

\subsection{Nested-loop design}

\subsubsection{Inner-loop equivalent plant}
\[
G_{\mathrm{inner,cl}}(s)=\frac{C_{\mathrm{inner}}(s)G_{\mathrm{inner}}(s)}
{1+C_{\mathrm{inner}}(s)G_{\mathrm{inner}}(s)}
\]
Design and verify the inner loop before treating it as part of the outer-loop plant.

\subsubsection{Outer-loop transfer}
\[
L_{\mathrm{outer}}(s)=C_{\mathrm{outer}}(s)G_{\mathrm{outer}}(s)
\]
The outer loop uses the stabilized inner closed-loop behavior as its plant model.

\subsection{Exam recipe: decide stability for an unstable open-loop plant}

\textbf{Use when:} the task combines a right-half-plane pole with \(K_P\), Bode, or Nyquist information.

\textbf{Steps:}
\begin{enumerate}
\item Count open-loop right-half-plane poles \(P\).
\item Use Nyquist encirclements or the characteristic polynomial, not the stable-plant Bode shortcut.
\item If using \(K_P\), scale the Nyquist curve or insert \(K_P\) in \(1+K_PG(s)=0\).
\item Require \(Z=0\) or all closed-loop poles in the left half-plane.
\end{enumerate}

\textbf{Fast checks:} a positive phase margin read from a stable-plant formula is not sufficient when \(P>0\).

\textbf{Common traps:} forgetting an open-loop pole at the origin or in the right half-plane.
