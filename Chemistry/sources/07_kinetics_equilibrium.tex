\section{Kinetics and Chemical Equilibrium}

\subsection{Ctrl+F keywords: rate law, reaction order, first order, second order, half-life, Arrhenius, catalyst, equilibrium, Le Chatelier, \(K_p\)}
Use for concentration-time data, identifying order, half-life, activation energy, catalyst statements, equilibrium expressions, \(Q\) versus \(K\), and gas equilibria.

\subsection{Kinetics formulas}

\subsubsection{Reaction rate from stoichiometry}
\[
a\ce{A}+b\ce{B}\rightarrow c\ce{C}+d\ce{D}
\]
\[
r=-\frac{1}{a}\frac{d[\ce{A}]}{dt}
=-\frac{1}{b}\frac{d[\ce{B}]}{dt}
=\frac{1}{c}\frac{d[\ce{C}]}{dt}
=\frac{1}{d}\frac{d[\ce{D}]}{dt}
\]
Divide by coefficients so every species reports the same reaction rate.

\subsubsection{Rate law and overall order}
\[
r=k[\ce{A}]^m[\ce{B}]^n, \qquad \text{overall order}=m+n
\]
Orders are found experimentally unless the problem states an elementary step.

\subsubsection{Integrated rate laws}
\[
\begin{array}{c c c}
\toprule
\text{order} & \text{linear plot} & \text{slope}\\
\midrule
0 & [\ce{A}]_t=-kt+[\ce{A}]_0 & -k\\
1 & \ln[\ce{A}]_t=-kt+\ln[\ce{A}]_0 & -k\\
2 & \dfrac{1}{[\ce{A}]_t}=kt+\dfrac{1}{[\ce{A}]_0} & k\\
\bottomrule
\end{array}
\]
Use the plot or transformed data that is linear to identify the order.

\subsubsection{Half-life}
\[
t_{1/2}^{(0)}=\frac{[\ce{A}]_0}{2k}, \qquad
t_{1/2}^{(1)}=\frac{\ln2}{k}, \qquad
t_{1/2}^{(2)}=\frac{1}{k[\ce{A}]_0}
\]
Only first-order half-life is independent of initial concentration.

\subsubsection{Arrhenius equation}
\[
k=Ae^{-E_{\mathrm a}/RT}, \qquad
\ln k=-\frac{E_{\mathrm a}}{R}\frac{1}{T}+\ln A
\]
A plot of \(\ln k\) versus \(1/T\) has slope \(-E_{\mathrm a}/R\).

\subsubsection{Two-temperature Arrhenius relation}
\[
\ln\left(\frac{k_2}{k_1}\right)
=-\frac{E_{\mathrm a}}{R}\left(\frac{1}{T_2}-\frac{1}{T_1}\right)
\]
Use kelvin. If \(T_2>T_1\), then usually \(k_2>k_1\).

\subsubsection{Catalyst effect}
\[
E_{\mathrm a,cat}<E_{\mathrm a,uncat}, \qquad K_{\mathrm{eq}}\ \text{unchanged}
\]
A catalyst speeds forward and reverse reactions by lowering activation barriers. It does not change \(\Delta G^\circ\) or the equilibrium constant.

\subsection{Exam recipe: determine reaction order from concentration-time data}
\textbf{Use when:} a table of concentration versus time is given, or answer options show first/second order.

\textbf{Steps:}
\begin{enumerate}
    \item Test which transformed quantity is linear: \([\ce{A}]\), \(\ln[\ce{A}]\), or \(1/[\ce{A}]\).
    \item Select zero, first, or second order from the linear plot.
    \item Compute \(k\) from the slope with the correct sign.
    \item Use the matching integrated law for prediction.
\end{enumerate}

\textbf{Fast checks:} first-order equal time intervals give equal fractional decrease; second-order \(1/[\ce{A}]\) increases linearly.

\subsection{Equilibrium formulas}

\subsubsection{Reaction quotient and equilibrium constant}
\[
\alpha\ce{A}+\beta\ce{B}\rightleftharpoons\gamma\ce{C}+\delta\ce{D}, \qquad
Q=\frac{a_{\ce{C}}^\gamma a_{\ce{D}}^\delta}{a_{\ce{A}}^\alpha a_{\ce{B}}^\beta}
\]
At equilibrium, \(Q=K\). Pure solids, pure liquids, and the solvent are omitted.

\subsubsection{Direction from \(Q\) and \(K\)}
\[
Q<K:\ \text{forward}, \qquad Q=K:\ \text{equilibrium}, \qquad Q>K:\ \text{reverse}
\]
The system shifts to make \(Q\) approach \(K\).

\subsubsection{Gas equilibrium \(K_p\)}
\[
K_p=\frac{(p_{\ce{C}})^\gamma(p_{\ce{D}})^\delta}{(p_{\ce{A}})^\alpha(p_{\ce{B}})^\beta}
\]
Use partial pressures with stoichiometric powers. Omit pure condensed phases.

\subsubsection{Relation between \(K_p\) and \(K_c\)}
\[
K_p=K_c(RT)^{\Delta n_{\mathrm{gas}}}
\]
\(\Delta n_{\mathrm{gas}}\) is gaseous product coefficients minus gaseous reactant coefficients.

\subsubsection{Le Chatelier shift rules}
\[
\text{disturbance}\Rightarrow\text{shift that reduces the disturbance}
\]
Adding reactant shifts toward products; removing product shifts toward products. Decreasing volume favors the side with fewer gas moles.

\subsection{Exam recipe: equilibrium expression and Le Chatelier}
\textbf{Use when:} the problem asks for \(K\), \(Q\), \(K_p\), direction of shift, or statement validity.

\textbf{Steps:}
\begin{enumerate}
    \item Write products over reactants with coefficients as exponents.
    \item Omit pure solids, pure liquids, and solvent.
    \item For gases use partial pressures for \(K_p\).
    \item Compare \(Q\) with \(K\) for direction, or apply the shift rule for disturbances.
\end{enumerate}

\textbf{Common traps:} including solids in \(K\); reversing \(Q<K\) direction; treating catalysts as changing equilibrium.
