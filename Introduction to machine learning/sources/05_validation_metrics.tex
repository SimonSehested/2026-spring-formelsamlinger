\section*{VALIDATION / METRICS / MODEL OUTPUT}
\subsection*{HOLD-OUT / K-FOLD / NESTED CV}
\[
\fid[eq:cv-error]\begin{aligned}
E_T&=\frac1{|T|}\sum_{i\in T}L(y_i,\hat f(x_i)),&
E_{\rm CV}&=\frac1K\sum_{k=1}^KE_k,\\
\hat E_{\rm gen}&=\frac1{K_1}\sum_{i=1}^{K_1}E^{\rm test}_{s_i^*,i}
\end{aligned}
\]
MEANS: $T$ = test/held-out set; $L$ = loss; $E_k$ = fold validation error; $K_1$ = outer folds; $s_i^*$ = model/complexity selected by inner CV inside outer fold $i$; $E^{\rm test}_{s_i^*,i}$ = outer test error of that selected model. CALC NESTED: inner folds choose model in each outer fold; outer test errors estimate generalization; average only the chosen model's outer errors.

\subsection*{CONFUSION MATRIX}
\[
\fid[eq:conf-mat]\begin{array}{c|cc}
&\hat y=+&\hat y=-\\\hline
y=+&TP&FN\\
y=-&FP&TN
\end{array}
\]
MEANS: $+$ = stated positive class; $TP$ = actual positive predicted positive; $FN$ = actual positive predicted negative; $FP$ = actual negative predicted positive; $TN$ = actual negative predicted negative. READ: do not trust row/column position blindly; identify actual class and predicted class from table labels.

\subsection*{CLASSIFICATION METRICS}
\[
\fid[eq:class-metrics]\begin{aligned}
\operatorname{Acc}&=\frac{TP+TN}{N},&
\operatorname{Err}&=1-\operatorname{Acc},\\
\operatorname{Recall}=\operatorname{TPR}&=\frac{TP}{TP+FN},&
\operatorname{FPR}&=\frac{FP}{FP+TN},\\
\operatorname{Precision}&=\frac{TP}{TP+FP},&
F_1&=\frac{2PR}{P+R}
\end{aligned}
\]
MEANS: $N=TP+FN+FP+TN$; $P$ = precision; $R$ = recall. READ: recall = fraction of actual positives found; precision = fraction of predicted positives correct; FPR = fraction of actual negatives wrongly predicted positive.

\subsection*{ROC / AUC / THRESHOLD}
\[
\fid[eq:roc]\begin{aligned}
\operatorname{ROC}(\tau)&=(\operatorname{FPR}(\tau),\operatorname{TPR}(\tau)),\\
\operatorname{AUC}&=\text{area under ROC}
\end{aligned}
\]
MEANS: threshold $\tau$ turns scores/probabilities into classes; ROC x-axis = FPR; ROC y-axis = TPR. CALC FROM SORTED SCORES: start at $(0,0)$ with no predicted positives; lowering threshold over a true positive moves up; over a true negative moves right. READ: upper-left is best; random ranking has AUC near 0.5; high threshold usually predicts fewer positives.
