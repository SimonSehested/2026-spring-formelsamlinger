\section*{Prediction and boundaries}
\subsection*{Linear and ridge regression}
\textbf{Purpose:} predict a numeric response as intercept plus weighted attributes.
\[
\fid[eq:lin-ridge]\begin{aligned}
\bar x&=[1,x_1,\ldots,x_M]^\top,\quad X_a=[\mathbf1,X],\\
\hat y&=\bar x^\top w,\\
w_{\rm LS}&=(X_a^\top X_a)^{-1}X_a^\top y,\\
D&=\operatorname{diag}(0,1,\ldots,1),\\
w_\lambda&=(X_a^\top X_a+\lambda D)^{-1}X_a^\top y.
\end{aligned}
\]
\[
\fid[eq:ridge-cost]E_\lambda(w)=\sum_i(y_i-\bar x_i^\top w)^2+
\lambda\sum_{j=1}^Mw_j^2 .
\]
\textbf{Symbols:} $w_0$ is intercept; $w_j$ is the change in $\hat y$ per unit $x_j$; $D$ leaves the intercept unpenalized; $\lambda$ controls shrinkage.
\textbf{Do this:} use \fref{eq:lin-ridge} to add the leading $1$, multiply each input by its weight, and add. For $\lambda$, compute validation/CV error for each candidate and choose the smallest; larger $\lambda$ in \fref{eq:ridge-cost} pulls non-intercept weights toward $0$.
\textbf{Read output:} if attributes are standardized, compare $|w_j|$ directly; otherwise coefficient size also depends on measurement units. Positive $w_j$ raises prediction when $x_j$ rises and all other inputs are fixed.

\subsection*{Logistic regression}
\[
\fid[eq:logistic]\begin{aligned}
a&=\bar x^\top w,&
\hat p&=P(y=1|x)=\sigma(a)=\frac1{1+e^{-a}},\\
\hat y&=\ind[\hat p\ge\tau],&
\log\frac{\hat p}{1-\hat p}&=\bar x^\top w .
\end{aligned}
\]
\textbf{How to read it:} $a$ is log-odds; positive $w_j$ raises positive-class probability when $x_j$ increases. At $\tau=0.5$, classify positive when $a\ge0$; the boundary is $\bar x^\top w=0$.
\textbf{Do this:} use \fref{eq:logistic}; insert the intercept $1$; compute $a$; convert with sigmoid; compare $\hat p$ with the stated threshold $\tau$ and the stated positive class.
\textbf{Boundary in 2D:} solve $w_0+w_1x_1+w_2x_2=0$ for one axis, e.g. $x_2=-(w_0+w_1x_1)/w_2$. Points on one side have $a>0$, the other side $a<0$.

\subsection*{K-nearest neighbours}
\[
\fid[eq:knn]\begin{aligned}
\hat y_{\rm reg}(x)&=\frac1K\sum_{i\in N_K(x)}y_i,\\
\hat y_{\rm class}(x)&=\argmax_c\sum_{i\in N_K(x)}\ind[y_i=c].
\end{aligned}
\]
\textbf{Symbols:} $N_K(x)$ is the $K$ training rows with smallest chosen distance to $x$.
\textbf{Do this:} compute distances with \fref{eq:cov-dist}; sort rows by distance; keep the first $K$; average their $y_i$ for regression or count class votes for classification. Small $K$ gives local irregular boundaries; large $K$ smooths.

\subsection*{Multiclass scores and boundary plots}
\[
\fid[eq:softmax]p_c=\frac{e^{a_c}}{\sum_je^{a_j}},\qquad \hat y=\argmax_cp_c .
\]
\textbf{Do this:} use \fref{eq:softmax}; exponentiate every class score, divide by the sum, then choose the largest probability. Straight boundary = linear score; axis-aligned regions = tree; local cells = KNN; smooth curved boundary = nonlinear features or neural network.
