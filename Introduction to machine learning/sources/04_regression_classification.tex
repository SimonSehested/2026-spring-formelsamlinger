\section*{PREDICTION / BOUNDARIES}
\subsection*{LINEAR / RIDGE REGRESSION}
\[
\fid[eq:lin-ridge]\begin{aligned}
\bar x&=[1,x_1,\ldots,x_M]^\top,&
\hat y&=\bar x^\top w,\\
w_{\rm LS}&=(X_a^\top X_a)^{-1}X_a^\top y,\\
w_\lambda&=(X_a^\top X_a+\lambda D)^{-1}X_a^\top y
\end{aligned}
\]
\[
\fid[eq:ridge-cost]\begin{aligned}
E_\lambda(w)&=\sum_i(y_i-\bar x_i^\top w)^2+
\lambda\sum_{j=1}^Mw_j^2,\\
D&=\operatorname{diag}(0,1,\ldots,1)
\end{aligned}
\]
MEANS: $\bar x$ = input with leading 1 for intercept; $X_a$ = data with intercept column; $w_0$ = intercept; $w_j$ = slope/weight; $\lambda$ = regularization strength; $D$ usually leaves intercept unpenalized. CALC: standardize test point if training used standardized data, insert leading 1, multiply by weights, add. READ: positive $w_j$ increases prediction when $x_j$ increases. REGULARIZATION: larger $\lambda$ shrinks weights, raises bias, usually lowers variance; more features/polynomials lower bias, often raise variance.

\subsection*{LOGISTIC REGRESSION / PROBABILITY}
\[
\fid[eq:logistic]\begin{aligned}
a&=\bar x^\top w,\\
\hat p&=P(y=1|x)=\sigma(a)=\frac1{1+e^{-a}},\\
\hat y&=\ind[\hat p\ge\tau],\quad
\log\frac{\hat p}{1-\hat p}=a
\end{aligned}
\]
MEANS: $a$ = score/log-odds; $\hat p$ = predicted probability of stated positive class; $\tau$ = threshold; $\sigma$ = sigmoid. READ: at $\tau=0.5$, boundary is $a=0$; positive $a$ gives $\hat p>0.5$. Positive weight raises positive-class probability, holding other inputs fixed.

\subsection*{FEATURE MAP / HEATMAP / BOUNDARY}
\[
\fid[eq:feature-map]\begin{aligned}
\tilde x&=\phi(x),\\
\hat y&=w^\top\tilde x\quad{\rm(regression)},\\
P(y=1|x)&=\sigma(w^\top\tilde x)\quad{\rm(classification)}
\end{aligned}
\]
MEANS: $\phi(x)$ = transformed features, e.g. $[1,x_1,x_2^2]$; $\tilde x$ = feature vector after transformation. READ HEATMAP: contours follow constant $w^\top\tilde x$; linear features give straight contours; squared/cubic/sine/cos features give curved/repeated contours; huge regularization gives nearly flat output.

\subsection*{K-NEAREST NEIGHBOURS}
\[
\fid[eq:knn]\begin{aligned}
\hat y_{\rm reg}(x)&=\frac1K\sum_{i\in N_K(x)}y_i,\\
\hat y_{\rm class}(x)&=\argmax_c\sum_{i\in N_K(x)}\ind[y_i=c]
\end{aligned}
\]
MEANS: $N_K(x)$ = $K$ training rows closest to test row $x$ using chosen distance; $c$ = class. CALC: compute distances, sort, keep $K$, average targets or count class votes. READ BOUNDARY: small $K$ = local/jagged; large $K$ = smoother. If tied classes, use the exam's stated tie rule; otherwise nearest among tied classes is common.
