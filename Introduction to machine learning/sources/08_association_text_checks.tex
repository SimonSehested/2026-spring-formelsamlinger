\section*{Associations and sparse vectors}
\subsection*{Association rules and Apriori}
\[
\fid[eq:assoc]\begin{aligned}
\operatorname{supp}(X)&=\frac{\#(X)}N,\\
\operatorname{conf}(X\to Y)&=\frac{\operatorname{supp}(X\cup Y)}{\operatorname{supp}(X)},\\
\operatorname{lift}(X\to Y)&=\frac{\operatorname{conf}(X\to Y)}{\operatorname{supp}(Y)}.
\end{aligned}
\]
\textbf{Symbols:} $X$ antecedent, $Y$ consequent, disjoint itemsets; $\#(X)$ is baskets containing $X$.
\textbf{Do this:} use \fref{eq:assoc}; count baskets containing $X$, $Y$, and $X\cup Y$; support is count divided by $N$; confidence is the fraction of $X$ baskets that also contain $Y$; lift compares confidence to baseline support of $Y$. Apriori: discard candidate itemsets below minimum support, and only extend retained frequent itemsets.

\subsection*{Binary attributes and Jaccard}
\[
\fid[eq:jaccard]\begin{gathered}
\begin{array}{c|cc}
&b=1&b=0\\\hline
a=1&f_{11}&f_{10}\\
a=0&f_{01}&f_{00}
\end{array}
\\
J(a,b)=\frac{f_{11}}{f_{11}+f_{10}+f_{01}},\qquad
\operatorname{SMC}=\frac{f_{11}+f_{00}}M.
\end{gathered}
\]
\textbf{Do this:} for each attribute, count which cell of \fref{eq:jaccard} the pair $(a_m,b_m)$ falls in. Jaccard ignores shared zeros, so use it for sparse baskets/word presence; SMC counts both shared ones and shared zeros.

\subsection*{Text vectors and cosine}
\[
\fid[eq:tfidf]\operatorname{tfidf}_{ij}=\operatorname{tf}_{ij}\log\frac{N}{\operatorname{df}_j},\qquad
\cos(x,y)=\frac{x^\top y}{\|x\|_2\|y\|_2}.
\]
\textbf{Do this:} use \fref{eq:tfidf}; multiply term frequency by inverse document frequency, so rare terms get larger weight; compare documents with cosine from \fref{eq:norm-cos}. Cosine near $1$ means similar word-weight direction, independent of document length scale.
