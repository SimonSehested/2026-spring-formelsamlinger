\section*{Associations and sparse similarity}
\subsection*{Association rules and Apriori}
\textbf{Purpose:} Quantify which items co-occur and form frequent rules.
\[
\operatorname{supp}(X)=\frac{\#(X)}N,
\]
\[
\operatorname{conf}(X\to Y)=\frac{\operatorname{supp}(X\cup Y)}
{\operatorname{supp}(X)},
\]
\[
\operatorname{lift}(X\to Y)=\frac{\operatorname{conf}(X\to Y)}
{\operatorname{supp}(Y)}.
\]
\textbf{Symbols:} $X$ = antecedent itemset; $Y$ = consequent itemset; $\#(X)$ = baskets containing $X$; $N$ = number of baskets; $\operatorname{supp}$ = support; $\operatorname{conf}$ = confidence; lift compares observed confidence with baseline support of $Y$. For a rule, $X,Y$ are disjoint.

\textbf{How to read a rule table:} support is prevalence of all rule items; confidence is conditional frequency of $Y$ among baskets with $X$; lift $>1$ indicates more co-occurrence than expected from $Y$'s frequency.

\textbf{Do this:} count $X$, $Y$ and $X\cup Y$; calculate support/confidence/lift; for Apriori, keep itemsets meeting support and extend only retained sets.

\subsection*{Binary attributes and Jaccard}
\[
\begin{array}{c|cc}
&b=1&b=0\\\hline
a=1&f_{11}&f_{10}\\
a=0&f_{01}&f_{00}
\end{array}
\qquad
J(a,b)=\frac{f_{11}}{f_{11}+f_{10}+f_{01}},
\]
\[
\operatorname{SMC}=\frac{f_{11}+f_{00}}M.
\]
\textbf{Symbols/means:} $a,b$ = two binary vectors; $f_{rs}$ = number of attributes having values $r$ in $a$ and $s$ in $b$; $M$ = attribute count; $J$ = Jaccard; SMC = simple matching coefficient. Jaccard ignores shared absences; SMC counts both agreements.

\textbf{Use when:} Jaccard suits sparse baskets or word-presence vectors where common zeros do not show similarity.

\subsection*{Text vectors and cosine}
\textbf{Purpose:} Compare documents by weighted word content rather than raw document length.
\[
\operatorname{tfidf}_{ij}=\operatorname{tf}_{ij}\log\frac{N}{\operatorname{df}_j},
\qquad
\cos(x,y)=\frac{x^\top y}{\|x\|_2\|y\|_2}.
\]
\textbf{Symbols:} $\operatorname{tf}_{ij}$ = term count/weight for word $j$ in document $i$; $\operatorname{df}_j$ = documents containing word $j$; $N$ = documents; $x,y$ = tf--idf document vectors; $x^\top y$ = dot product; $\|\cdot\|_2$ = vector length.

\textbf{How to read it:} rare terms receive larger inverse-document-frequency weight; cosine near $1$ means documents point in similar word-weight directions.
