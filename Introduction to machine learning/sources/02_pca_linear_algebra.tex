\section*{PCA and linear algebra}
\subsection*{SVD/PCA notation}
For centered or standardized $\tilde X$:
\[
\tilde X=U S V^\top,\quad V=[v_1,\ldots,v_M],\quad Z=\tilde X V=US .
\]
Projection of observation $\tilde x_i$ on first $K$ PCs:
\[
z_i=\tilde x_i V_K,\qquad \hat x_i=z_iV_K^\top+\bar x .
\]
\note{Trap} Exam matrices often print $V$; PC $k$ is column $k$, not row $k$. Sign of a PC is arbitrary, but projections and loadings must use the same sign.

\subsection*{Explained variance}
\[
\lambda_k=\frac{s_k^2}{N-1},\quad
\text{EVR}_k=\frac{s_k^2}{\sum_j s_j^2},\quad
\text{cumEVR}(K)=\frac{\sum_{k=1}^K s_k^2}{\sum_j s_j^2}.
\]
\note{Use} Choose fewest $K$ with cumulative explained variance above target (often $90\%$).
\note{Check} If standardized, total variance $\approx M$ and $\|\tilde X\|_F^2=\sum_k s_k^2\approx (N-1)M$.

\subsection*{Reading PC directions}
\[
z_{ik}=\tilde x_i^\top v_k=\sum_m \tilde x_{im}v_{mk}.
\]
\note{Recipe} To evaluate a statement: identify signs/magnitudes of $\tilde x_m$, multiply by column $v_k$, sum dominant terms, compare sign.
\note{Trap} ``High original value'' means positive standardized value only if above training mean.

\subsection*{Common linear algebra checks}
\[
\|x\|_2=\sqrt{x^\top x},\quad
\|X\|_F^2=\sum_{ij}x_{ij}^2=\sum_k s_k^2,\quad
\cos(x,y)=\frac{x^\top y}{\|x\|\|y\|}.
\]
\note{Trap} Cosine ignores magnitude; Euclidean distance does not. Sparse text vectors often compare with cosine or Jaccard, not raw Euclidean distance.
