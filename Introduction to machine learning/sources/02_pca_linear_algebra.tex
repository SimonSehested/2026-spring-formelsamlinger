\section*{PCA and SVD}
\subsection*{Components, scores and reconstruction}
\textbf{Purpose:} Replace correlated attributes by orthogonal directions ordered by variation.
\[
\tilde X=USV^\top,\qquad V=[v_1,\ldots,v_M],
\]
\[
Z=\tilde XV=US,\qquad z_{ik}=\tilde x_i^\top v_k.
\]
\[
z_i=\tilde x_iV_K,\qquad \hat{\tilde x}_i=z_iV_K^\top .
\]
\textbf{Symbols:} $U$ = left singular-vector matrix; $S$ = diagonal singular-value matrix; $V$ = loading directions; column $v_k$ is PC $k$; $Z$ = score matrix; $z_i$ = PC coordinates of row $i$; $V_K$ retains $K$ PCs; $\hat{\tilde x}_i$ = reconstructed standardized row; $s_k$ = singular value.

\textbf{How to read it:} large $|v_{mk}|$ means attribute $m$ defines PC $k$ strongly. A positive score lies in the direction of positive loadings. Reversing all signs on one PC changes neither the PCA representation nor distances.

\subsection*{Explained variance}
\[
\lambda_k=\frac{s_k^2}{N-1},\quad
\operatorname{EVR}_k=\frac{s_k^2}{\sum_j s_j^2},\quad
\operatorname{cumEVR}(K)=\frac{\sum_{k=1}^Ks_k^2}{\sum_js_j^2}.
\]
\textbf{Symbols/means:} $\lambda_k$ = variance along PC $k$; $\operatorname{EVR}_k$ = its fraction of total variance; $\operatorname{cumEVR}(K)$ = fraction retained by the first $K$ PCs.

\textbf{Do this:} square singular values; divide by their total; accumulate in PC order; choose the smallest $K$ reaching the target.

\subsection*{Read PCA output and plots}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\textbf{Object} & \textbf{Interpretation}\\ \midrule
Loading column $v_k$ & Read down the column: sign and magnitude connect original attributes to PC $k$.\\
Score plot & Each point is one observation; nearby points have similar retained attribute profiles.\\
Loading plot & Arrows/points are attributes; similar direction means similar contribution pattern.\\
Biplot & Observations projected along a variable direction tend to have high values of that variable.\\
\bottomrule
\end{tabularx}
\textbf{Do this for a projection:} take the requested column of $V$; multiply corresponding standardized attribute values; add terms to obtain $z_{ik}$.

\subsection*{Norm and cosine}
\[
\|x\|_2=\sqrt{x^\top x},\qquad
\|X\|_F^2=\sum_{ij}x_{ij}^2=\sum_ks_k^2,
\]
\[
\cos(x,y)=\frac{x^\top y}{\|x\|\|y\|}.
\]
\textbf{Symbols/means:} $\|\cdot\|_2$ = vector length; $\|X\|_F$ = Frobenius length of all matrix entries; $x^\top y$ = dot product; $\cos(x,y)$ reads similar direction, while Euclidean distance also responds to magnitude.
