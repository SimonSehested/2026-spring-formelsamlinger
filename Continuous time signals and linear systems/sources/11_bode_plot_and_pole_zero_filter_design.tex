\section{Bode Plot and Pole-Zero Filter Design}

\subsection{Pole-Zero Interpretation}

\subsubsection{Frequency response from pole-zero distances}
\[
|H(j\omega)|=|K|\frac{\prod_i|j\omega-z_i|}{\prod_k|j\omega-p_k|},
\qquad
\angle H(j\omega)=\angle K+\sum_i\angle(j\omega-z_i)-\sum_k\angle(j\omega-p_k).
\]
Magnitude is the ratio of distances from \(j\omega\) to zeros and poles. Phase is the sum of zero angles minus pole angles.

\subsubsection{Filter type from zeros at the origin}
\[
H(s)\propto s^m \quad \Rightarrow \quad H(0)=0.
\]
Zeros at the origin reject DC and are characteristic of highpass and bandpass transfer functions.

\subsubsection{Notch filter zeros}
\[
H_{\mathrm{notch}}(s)=K\frac{s^2+\omega_0^2}{s^2+\frac{\omega_0}{Q}s+\omega_0^2}.
\]
Zeros on the imaginary axis at \(\pm j\omega_0\) reject the notch frequency.

\subsection{Frequency Transformations}

\subsubsection{Lowpass frequency scaling}
\[
H_{\mathrm{LP}}(s)=H_n\!\left(\frac{s}{\omega_c}\right).
\]
Frequency scaling moves a normalized cutoff from \(1\ \mathrm{rad/s}\) to \(\omega_c\).

\subsubsection{Lowpass to highpass substitution}
\[
s_n=\frac{\omega_c}{s}.
\]
Substituting \(s_n=\omega_c/s\) in a normalized lowpass transfer function gives a highpass transfer function with cutoff \(\omega_c\).

\subsubsection{Lowpass to bandpass substitution}
\[
s_n=\frac{s^2+\omega_0^2}{B s},
\qquad B=\omega_2-\omega_1,\qquad \omega_0=\sqrt{\omega_1\omega_2}.
\]
The lowpass-to-bandpass transformation maps a normalized prototype to passband edges \(\omega_1\) and \(\omega_2\).

\subsubsection{Cascade of filter stages}
\[
H(s)=\prod_{k=1}^{M}H_k(s).
\]
Higher-order filters are implemented as cascades of first-order and second-order stages. Bode magnitudes in dB and phases add across stages.

\subsection{Tracking Filters}

\subsubsection{Tracking-filter numerator choice}
\[
T(s)=\frac{as+b}{(s+p_1)(s+p_2)}.
\]
The numerator coefficients place a zero and set the gain. Exam tasks use this form to match specified gain or phase behavior at selected frequencies.

\subsubsection{Solving gain constraints}
\[
T(0)=\frac{b}{p_1p_2},\qquad
\lim_{s\to\infty}sT(s)=a.
\]
Low-frequency and high-frequency asymptotes provide direct algebraic constraints on numerator coefficients.
